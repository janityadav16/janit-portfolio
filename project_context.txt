    });
  }, { threshold: 0.2 });
  const socGrid = document.querySelector('.contact-social-grid');
  if (socGrid) contactSocialObs.observe(socGrid);

  // --- FOOTER ANIMATIONS ---
  const footerObs = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (!entry.isIntersecting) return;
      const brand = entry.target.querySelector('.footer-brand');
      const links = entry.target.querySelector('.footer-links');
      const socials = entry.target.querySelector('.footer-socials');
      const copyright = entry.target.querySelector('.footer-bottom p');

      [brand, links, socials].forEach((el, i) => {
        if (!el) return;
        el.style.opacity = '0';
        el.style.transform = 'translateY(20px)';
        el.style.transition = `opacity 0.5s ${i * 0.15}s, transform 0.5s ${i * 0.15}s`;
        requestAnimationFrame(() => { el.style.opacity = '1'; el.style.transform = 'translateY(0)'; });
      });

      // Copyright typing
      if (copyright) {
        const txt = copyright.textContent;
        copyright.textContent = '';
        let ci = 0;
        setTimeout(() => {
          function typeCopy() {
            if (ci < txt.length) { copyright.textContent += txt[ci]; ci++; setTimeout(typeCopy, 20); }
          }
          typeCopy();
        }, 500);
      }

      footerObs.unobserve(entry.target);
    });
  }, { threshold: 0.2 });
  const footerEl = document.querySelector('.footer');
  if (footerEl) footerObs.observe(footerEl);

  // Footer social icons 360 rotation on hover
  document.querySelectorAll('.footer-social-links .social-icon').forEach(icon => {
    icon.style.transition = 'transform 0.4s ease';
    icon.addEventListener('mouseenter', () => { icon.style.transform = 'rotate(360deg)'; });
    icon.addEventListener('mouseleave', () => { icon.style.transform = 'rotate(0deg)'; });
  });

  // --- SKILL CARD PILL TAGS STAGGER ---
  const skillPillObs = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (!entry.isIntersecting) return;
      const pills = entry.target.querySelectorAll('.skill-name, .skill-proficiency');
      pills.forEach((pill, i) => {
        pill.style.opacity = '0';
        pill.style.transform = 'scale(0.5)';
        pill.style.transition = `opacity 0.3s ${i * 0.03}s, transform 0.3s cubic-bezier(0.34,1.56,0.64,1) ${i * 0.03}s`;
        requestAnimationFrame(() => { pill.style.opacity = '1'; pill.style.transform = 'scale(1)'; });
      });
      skillPillObs.unobserve(entry.target);
    });
  }, { threshold: 0.3 });
  document.querySelectorAll('.skill-card').forEach(c => skillPillObs.observe(c));

  // --- PROJECT tags typewriter stagger ---
  const projTagObs = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (!entry.isIntersecting) return;
      const tags = entry.target.querySelectorAll('.project-tag');
      tags.forEach((tag, i) => {
        tag.style.opacity = '0';
        tag.style.transform = 'translateX(15px)';
        tag.style.transition = `opacity 0.3s ${i * 0.05}s, transform 0.3s ${i * 0.05}s`;
        requestAnimationFrame(() => { tag.style.opacity = '1'; tag.style.transform = 'translateX(0)'; });
      });
      // Feature bullets slide in
      const features = entry.target.querySelectorAll('.feature-row');
      features.forEach((f, i) => {
        f.style.opacity = '0';
        f.style.transform = 'translateX(-15px)';
        f.style.transition = `opacity 0.3s ${0.2 + i * 0.08}s, transform 0.3s ${0.2 + i * 0.08}s`;
        requestAnimationFrame(() => { f.style.opacity = '1'; f.style.transform = 'translateX(0)'; });
      });
      projTagObs.unobserve(entry.target);
    });
  }, { threshold: 0.25 });
  document.querySelectorAll('.project-card').forEach(c => projTagObs.observe(c));

});
